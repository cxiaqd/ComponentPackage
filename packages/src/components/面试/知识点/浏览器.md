### 问题3：浏览器的事件循环机制是怎样的？宏任务和微任务有什么区别？

**考察点**：对事件循环的理解

**参考答案**：

**事件循环流程**：

```javascript
// 执行流程
1. 执行同步代码（当前调用栈）
2. 清空微任务队列（Promise.then, MutationObserver, queueMicrotask）
3. 执行一个宏任务（setTimeout, setInterval, I/O, UI渲染）
4. 重复步骤 2-3

// 任务类型
宏任务（MacroTask）：setTimeout, setInterval, setImmediate, I/O, UI渲染, MessageChannel
微任务（MicroTask）：Promise.then, MutationObserver, queueMicrotask, process.nextTick(Node)
```

**复杂示例**：

```javascript
console.log('1');

setTimeout(() => {
  console.log('2');
  Promise.resolve().then(() => {
    console.log('3');
  });
}, 0);

Promise.resolve().then(() => {
  console.log('4');
  setTimeout(() => {
    console.log('5');
  }, 0);
});

console.log('6');

// 输出顺序：1, 6, 4, 2, 3, 5
// 解析：
// 同步：1, 6
// 微任务：4
// 宏任务：2（执行时产生微任务3）
// 微任务：3
// 宏任务：5
```

**实际应用**：

```javascript
// 1. 优化性能：批量 DOM 更新
const batchUpdates = [];
let pending = false;

function updateElement(el, props) {
  batchUpdates.push({ el, props });
  
  if (!pending) {
    pending = true;
    queueMicrotask(() => {
      batchUpdates.forEach(({ el, props }) => {
        Object.assign(el.style, props);
      });
      batchUpdates.length = 0;
      pending = false;
    });
  }
}

// 2. 模拟 requestIdleCallback
function requestIdleCallbackPolyfill(callback) {
  let start = Date.now();
  return setTimeout(() => {
    callback({
      didTimeout: false,
      timeRemaining: () => Math.max(0, 50 - (Date.now() - start))
    });
  }, 1);
}

// 3. 实现异步任务队列
class AsyncQueue {
  constructor() {
    this.queue = [];
    this.processing = false;
  }
  
  add(task) {
    this.queue.push(task);
    this.process();
  }
  
  async process() {
    if (this.processing) return;
    this.processing = true;
    
    while (this.queue.length) {
      const task = this.queue.shift();
      await task(); // 宏任务
      await Promise.resolve(); // 让出主线程
    }
    
    this.processing = false;
  }
}
```

### 问题4：什么是重排（Reflow）和重绘（Repaint）？如何避免？

**考察点**：对渲染机制的理解

**参考答案**：

**定义**：

```javascript
// 重排（Layout）：计算元素几何位置（宽高、位置）
// 触发时机：DOM 增删改、元素尺寸/位置变化、读取 offsetHeight 等属性

// 重绘（Paint）：填充像素（颜色、背景、可见性）
// 触发时机：改变颜色、背景、阴影等不影响布局的属性

// 合成（Composite）：将图层合并
// 触发时机：transform、opacity 动画
```

**避免重排重绘**：

```javascript
// ❌ 错误：多次触发重排
element.style.width = '100px';
element.style.height = '100px';
element.style.margin = '10px';

// ✅ 正确：批量修改
element.style.cssText = 'width:100px;height:100px;margin:10px;';
// 或使用 class
element.classList.add('new-size');

// ❌ 错误：强制同步布局
const width = element.offsetWidth; // 读取，触发 Layout
element.style.width = width + 10 + 'px'; // 再次触发 Layout

// ✅ 正确：批量读取，批量修改
const width = element.offsetWidth;
const height = element.offsetHeight;
// 所有读取完成后，统一修改
element.style.width = width + 10 + 'px';
element.style.height = height + 10 + 'px';

// ❌ 错误：动画改变 width/height
function animate() {
  element.style.width = x + 'px';
  element.style.height = y + 'px';
  requestAnimationFrame(animate);
}

// ✅ 正确：使用 transform
function animate() {
  element.style.transform = `translate(${x}px, ${y}px)`;
  requestAnimationFrame(animate);
}

// 使用 will-change 提示浏览器
.element {
  will-change: transform, opacity;
}
```

### 问题8：HTTP 缓存机制是怎样的？强缓存和协商缓存有什么区别？

**考察点**：对缓存策略的理解

**参考答案**：

**缓存流程**：

```
请求资源 → 强缓存是否命中 → 是 → 从缓存读取（200 from disk/memory cache）
                ↓ 否
         协商缓存是否命中 → 是 → 返回 304
                ↓ 否
         重新请求资源（200）
```

**强缓存**：

```javascript
// Cache-Control（优先级高于 Expires）
// 响应头
Cache-Control: max-age=31536000  // 缓存 1 年
Cache-Control: no-cache          // 每次使用缓存前验证
Cache-Control: no-store          // 完全不缓存
Cache-Control: public            // 可被任何缓存
Cache-Control: private           // 仅用户浏览器缓存

// Expires（HTTP/1.0）
Expires: Wed, 21 Oct 2025 07:28:00 GMT
```

**协商缓存**：

```javascript
// ETag（优先级高于 Last-Modified）
// 响应头
ETag: "33a64df551425fcc55e4d42a148795d9f25f89d4"

// 请求头
If-None-Match: "33a64df551425fcc55e4d42a148795d9f25f89d4"
// 匹配则返回 304

// Last-Modified
// 响应头
Last-Modified: Wed, 21 Oct 2024 07:28:00 GMT

// 请求头
If-Modified-Since: Wed, 21 Oct 2024 07:28:00 GMT
```

**实际应用**：

```javascript

    add_header Cache-Control "public, immutable";
}

    add_header Cache-Control "no-cache";
}

// 2. 前端强制刷新
// 使用版本号或 hash

<script src="app.a1b2c3d4.js"></script>
```

### 问题11：HTTP 状态码的含义和应用场景？

**考察点**：对 HTTP 协议的理解

**参考答案**：

**状态码分类**：

```javascript
// 1xx: 信息响应
100 Continue: 请求已接收，继续发送
101 Switching Protocols: 切换协议（WebSocket）

// 2xx: 成功
200 OK: 请求成功
201 Created: 资源创建成功
204 No Content: 请求成功，无返回内容

// 3xx: 重定向
301 Moved Permanently: 永久重定向
302 Found: 临时重定向
304 Not Modified: 缓存未修改
307 Temporary Redirect: 临时重定向（保持方法）

// 4xx: 客户端错误
400 Bad Request: 请求参数错误
401 Unauthorized: 未授权
403 Forbidden: 禁止访问
```

```javascript
  '/',
  '/styles.css',

      );
    })
  );

  event.respondWith(
    caches.match(event.request)
      .then(response => {
        // 缓存命中，返回缓存
        if (response) {
          return response;
        }
        
        // 网络请求
        return fetch(event.request)
          .then(response => {
            // 缓存动态请求
            const responseToCache = response.clone();
            caches.open(CACHE_NAME)
              .then(cache => {
                cache.put(event.request, responseToCache);
              });
            return response;
          })
          .catch(() => {
            // 离线时返回备用页面
            if (event.request.mode === 'navigate') {
```

**实现心跳保活**：

```javascript
class WebSocketClient {
  constructor(url, options = {}) {
    this.url = url;
    this.heartbeatInterval = options.heartbeatInterval || 30000;
    this.reconnectDelay = options.reconnectDelay || 3000;
    this.maxReconnectAttempts = options.maxReconnectAttempts || 5;
    this.reconnectAttempts = 0;
    this.isClosed = false;
    this.connect();
  }
  
  connect() {
    this.ws = new WebSocket(this.url);
    
    this.ws.onopen = () => {
      console.log('WebSocket 连接成功');
      this.reconnectAttempts = 0;
      this.startHeartbeat();
    };
    
    this.ws.onmessage = (event) => {
      this.handleMessage(event.data);
      this.resetHeartbeat(); // 收到消息，重置心跳
    };
    
    this.ws.onclose = () => {
      console.log('WebSocket 连接关闭');
      this.stopHeartbeat();
      if (!this.isClosed) {
        this.reconnect();
      }
    };
    
    this.ws.onerror = (error) => {
      console.error('WebSocket 错误:', error);
    };
  }
  
  startHeartbeat() {
    this.heartbeatTimer = setInterval(() => {
      if (this.ws.readyState === WebSocket.OPEN) {
        this.send('ping');
        
        // 设置超时检测
        this.pongTimeout = setTimeout(() => {
          console.log('心跳超时，重新连接');
          this.ws.close();
        }, 5000);
      }
    }, this.heartbeatInterval);
  }
  
  resetHeartbeat() {
    if (this.pongTimeout) {
      clearTimeout(this.pongTimeout);
    }
  }
  
  stopHeartbeat() {
    if (this.heartbeatTimer) {
      clearInterval(this.heartbeatTimer);
    }
    if (this.pongTimeout) {
      clearTimeout(this.pongTimeout);
    }
  }
  
  send(data) {
    if (this.ws.readyState === WebSocket.OPEN) {
      this.ws.send(data);
    }
  }
  
  handleMessage(data) {
    if (data === 'pong') {
      // 心跳响应
      this.resetHeartbeat();
    } else {
      // 业务消息
      console.log('收到消息:', data);
    }
  }
  
  reconnect() {
    if (this.reconnectAttempts < this.maxReconnectAttempts) {
      this.reconnectAttempts++;
      console.log(`第 ${this.reconnectAttempts} 次重连`);
      setTimeout(() => {
        this.connect();
      }, this.reconnectDelay * Math.pow(2, this.reconnectAttempts - 1));
    }
  }
  
  close() {
    this.isClosed = true;
    this.stopHeartbeat();
    this.ws.close();
  }
}
```
