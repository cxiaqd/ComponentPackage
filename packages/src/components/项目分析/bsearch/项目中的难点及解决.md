## 一、技术难点

### 1.1 大数据量渲染性能问题
s
**难点描述：**
```
┌─────────────────────────────────────────────────────────────┐
│                    问题场景                                  │
└─────────────────────────────────────────────────────────────┘

1. 搜索结果页面可能返回上万条数据
2. 每条数据包含多张图片（人脸、人体、车辆）
3. 同时渲染导致页面卡顿，甚至浏览器崩溃
4. 滚动时性能严重下降
```

**问题代码示例：**
```javascript

<template>
  <div>
    <div v-for="item in searchResults" :key="item.id">
      <img :src="item.faceUrl" />
      <img :src="item.bodyUrl" />
      <img :src="item.vehicleUrl" />
    </div>
  </div>
</template>

// 问题：searchResults 可能有 10000+ 条
// 每条 3 张图片，总共 30000+ 张图片同时加载
```

**解决方案：**

#### 方案1: 虚拟滚动

```javascript
// mixins/virtualScroll.js
export default {
  data() {
    return {
      visibleStart: 0,
      visibleEnd: 50,
      itemHeight: 120,
      containerHeight: 0
    };
  },
  computed: {
    visibleItems() {
      return this.searchResults.slice(this.visibleStart, this.visibleEnd);
    },
    totalHeight() {
      return this.searchResults.length * this.itemHeight;
    }
  },
  methods: {
    handleScroll(e) {
      const scrollTop = e.target.scrollTop;
      this.visibleStart = Math.floor(scrollTop / this.itemHeight);
      this.visibleEnd = this.visibleStart + Math.ceil(this.containerHeight / this.itemHeight) + 10;
    }
  }
};
```

#### 方案2: 分页加载

```javascript

  data() {
    return {
      currentPage: 1,
      pageSize: 50,
      loading: false,
      hasMore: true
    };
  },
  methods: {
    async loadMore() {
      if (this.loading || !this.hasMore) return;
      
      this.loading = true;
      try {
        const result = await this.fetchData({
          page: this.currentPage,
          pageSize: this.pageSize
        });
        
        this.searchResults = [...this.searchResults, ...result.data.list];
        this.hasMore = result.data.hasMore;
        this.currentPage++;
      } finally {
        this.loading = false;
      }
    }
  }
};
```

#### 方案3: 图片懒加载

```javascript

// utils/permission.js
import store from '@/store';

export class PermissionChecker {
  constructor() {
    this.store = store;
  }
  
  /**
   * 检查单个权限
   */
  has(permission) {
    const { userRole, moduleSale } = this.store.state;
    
    // 管理员拥有所有权限
    if (userRole.isAdmin) {
      return true;
    }
    
    // 销售项权限检查
    const saleMap = {
      'face:search': moduleSale.faceSale && !userRole.userRoleMap?.noface,
      'body:search': moduleSale.bodySale && !userRole.userRoleMap?.nobody,
      'vehicle:search': moduleSale.vehicleSale && !userRole.userRoleMap?.novehicle
    };
    
    return saleMap[permission] === true;
  }
  
  /**
   * 检查是否有任一权限
   */
  hasAny(permissions) {
    return permissions.some(p => this.has(p));
  }
  
  /**
   * 检查是否有所有权限
   */
  hasAll(permissions) {
    return permissions.every(p => this.has(p));
  }
}

export const permissionChecker = new PermissionChecker();

// utils/imageHelper.js

export class ImageHelper {
  /**
   * Base64 转 File
   */
  static base64ToFile(dataurl, filename = 'file') {
    const arr = dataurl.split(',');
    const mime = arr[0].match(/:(.*?);/)[1];
    const suffix = mime.split('/')[1];
    const bstr = atob(arr[1]);
    const u8arr = new Uint8Array(bstr.length);
    for (let i = 0; i < bstr.length; i++) {
      u8arr[i] = bstr.charCodeAt(i);
    }
    return new File([u8arr], `${filename}.${suffix}`, { type: mime });
  }
  
  /**
   * 生成裁剪 URL
   */
  static getCropUrl(originalUrl, rect) {
    if (!rect) return originalUrl;
    
    const img = new Image();
    img.src = originalUrl;
    
    return new Promise((resolve) => {
      img.onload = () => {
        const { width, height } = img;
        const x = Math.round(width * rect.x);
        const y = Math.round(height * rect.y);
        const w = Math.round(width * rect.width);
        const h = Math.round(height * rect.height);
        
        const cropParams = `${x}:${y}:${w}:${h}`;
        const encoded = btoa(`${atob(originalUrl.split('=')[1])}&act=get&ztype=4&zvalue=${cropParams}`);
        resolve(`/ngx/proxy?i=${encoded}`);
      };
    });
  }
  
  /**
   * 上传图片
   */
  static async uploadImage(file, options = {}) {
    const formData = new FormData();
    formData.append('file', file);
    
    const response = await http.upload({
      url: '/img/uploadPicture.do',
      data: formData
    });
    
    return response.data;
  }
  
  /**
   * 压缩图片
   */
  static async compressImage(file, quality = 0.8) {
    return new Promise((resolve) => {
      const reader = new FileReader();
      reader.onload = (e) => {
        const img = new Image();
        img.onload = () => {
          const canvas = document.createElement('canvas');
          const ctx = canvas.getContext('2d');
          canvas.width = img.width;
          canvas.height = img.height;
          ctx.drawImage(img, 0, 0);
          
          canvas.toBlob((blob) => {
            resolve(new File([blob], file.name, { type: blob.type }));
          }, 'image/jpeg', quality);
        };
        img.src = e.target.result;
      };
      reader.readAsDataURL(file);
    });
  }
}
```

### 2.1 多类型搜索结果融合

**难点描述：**
```

// 问题：融合逻辑复杂，难以维护
async function search(params) {
  const results = {
    face: [],
    body: [],
    vehicle: [],
    nonvehicle: []
  };
  
  // 并发搜索
  const [faceRes, bodyRes, vehicleRes] = await Promise.all([
    searchFace(params),
    searchBody(params),
    searchVehicle(params)
  ]);
  
  results.face = faceRes.data.list;
  results.body = bodyRes.data.list;
  results.vehicle = vehicleRes.data.list;
  
  // 融合排序
  const allResults = [
    ...results.face.map(item => ({ ...item, type: 'face' })),
    ...results.body.map(item => ({ ...item, type: 'body' })),
    ...results.vehicle.map(item => ({ ...item, type: 'vehicle' }))
  ];
  
  // 按时间排序
  allResults.sort((a, b) => b.captureTime - a.captureTime);
  
  return allResults;
}
```

**解决方案：**

#### 方案1: 融合结果处理器

```javascript
// utils/fusionHandler.js

export class FusionResultHandler {
  /**
   * 处理融合结果
   */
  static async handleFusionSearch(params) {
    const searchTypes = this.getSearchTypes(params);
    
    // 并发搜索
    const searchPromises = searchTypes.map(type => 
      this.searchByType(type, params)
    );
    
    const results = await Promise.allSettled(searchPromises);
    
    // 提取成功的结果
    const validResults = results
      .filter(r => r.status === 'fulfilled')
      .map(r => r.value);
    
    // 融合结果
    return this.fusionResults(validResults, params);
  }
  
  /**
   * 根据参数获取搜索类型
   */
  static getSearchTypes(params) {
    const types = [];
    if (params.modelTypes?.includes('face')) types.push('face');
    if (params.modelTypes?.includes('body')) types.push('body');
    if (params.modelTypes?.includes('vehicle')) types.push('vehicle');
    if (params.modelTypes?.includes('nonvehicle')) types.push('nonvehicle');
    return types;
  }
  
  /**
   * 按类型搜索
   */
  static async searchByType(type, params) {
    const apiMap = {
      face: api.searchFace,
      body: api.searchBody,
      vehicle: api.searchVehicle,
      nonvehicle: api.searchNonvehicle
    };
    
    const result = await apiMap[type](params);
    return {
      type,
      data: result.data.list,
      total: result.data.total
    };
  }
  
  /**
   * 融合结果
   */
  static fusionResults(results, params) {
    // 按时间融合
    const timeMap = new Map();
    
    results.forEach(({ type, data }) => {
      data.forEach(item => {
        const key = item.captureTime;
        if (!timeMap.has(key)) {
          timeMap.set(key, {
            captureTime: key,
            items: []
          });
        }
        timeMap.get(key).items.push({ ...item, type });
      });
    });
    
    // 转换为数组并排序
    const fusedData = Array.from(timeMap.values())
      .sort((a, b) => b.captureTime - a.captureTime);
    
    return {
      list: fusedData,
      total: fusedData.length
    };
  }
}
```

**优化效果：**
- 搜索逻辑清晰
- 易于扩展新类型
- 错误处理完善

#### 方案1: 表单验证工具类

```javascript
// utils/formValidator.js

export class FormValidator {
  constructor() {
    this.rules = new Map();
  }
  
  /**
   * 添加验证规则
   */
  addRule(field, rule) {
    if (!this.rules.has(field)) {
      this.rules.set(field, []);
    }
    this.rules.get(field).push(rule);
  }
  
  /**
   * 验证表单
   */
  async validate(data) {
    const errors = {};
    
    for (const [field, rules] of this.rules.entries()) {
      const value = data[field];
      
      for (const rule of rules) {
        const result = await this.validateRule(value, rule, data);
        if (result !== true) {
          errors[field] = result;
          break;
        }
      }
    }
    
    return {
      valid: Object.keys(errors).length === 0,
      errors
    };
  }
  
  /**
   * 验证单个规则
   */
  async validateRule(value, rule, formData) {
    const { required, validator, message } = rule;
    
    // 必填验证
    if (required && (value === undefined || value === null || value === '')) {
      return message || `${rule.field}不能为空`;
    }
    
    // 自定义验证器
    if (validator) {
      const result = await validator(value, formData);
      if (result !== true) {
        return result || message;
      }
    }
    
    return true;
  }
}

// 使用
const validator = new FormValidator();

validator.addRule('keyword', {
  field: '关键词',
  required: true,
  message: '请输入关键词'
});

validator.addRule('startTime', {
  field: '开始时间',
  required: true,
  validator: (value, formData) => {
    if (!value) return '请选择开始时间';
    if (formData.endTime && new Date(value) > new Date(formData.endTime)) {
      return '开始时间不能大于结束时间';
    }
    return true;
  }
});

const { valid, errors } = await validator.validate(formData);
```

// mixins/resourceManager.js

export default {
  data() {
    return {
      _timers: new Set(),
      _eventListeners: new Map(),
      _sockets: new Set(),
      _intervals: new Set()
    };
  },
  
  methods: {
    /**
     * 设置定时器
     */
    $setTimeout(callback, delay) {
      const timer = setTimeout(() => {
        callback();
        this._timers.delete(timer);
      }, delay);
      this._timers.add(timer);
      return timer;
    },
    
    /**
     * 设置间隔
     */
    $setInterval(callback, delay) {
      const timer = setInterval(callback, delay);
      this._intervals.add(timer);
      return timer;
    },
    
    /**
     * 添加事件监听
     */
    $addEventListener(target, event, handler, options) {
      target.addEventListener(event, handler, options);
      
      if (!this._eventListeners.has(target)) {
        this._eventListeners.set(target, []);
      }
      this._eventListeners.get(target).push({ event, handler, options });
    },
    
    /**
     * 创建 WebSocket
     */
    $createWebSocket(url) {
      const socket = new WebSocket(url);
      this._sockets.add(socket);
      return socket;
    },
    
    /**
     * 清理所有资源
     */
    $cleanupResources() {
      // 清理定时器
      this._timers.forEach(timer => clearTimeout(timer));
      this._timers.clear();
      
      // 清理间隔
      this._intervals.forEach(timer => clearInterval(timer));
      this._intervals.clear();
      
      // 清理事件监听
      this._eventListeners.forEach((listeners, target) => {
        listeners.forEach(({ event, handler, options }) => {
          target.removeEventListener(event, handler, options);
        });
      });
      this._eventListeners.clear();
      
      // 关闭 WebSocket
      this._sockets.forEach(socket => {
        if (socket.readyState === WebSocket.OPEN) {
          socket.close();
        }
      });
      this._sockets.clear();
    }
  },
  
  beforeDestroy() {
    this.$cleanupResources();
  }
};

// 使用
export default {
  mixins: [resourceManager],
  
  mounted() {
    this.$setInterval(() => {
      this.fetchData();
    }, 5000);
    
    this.$addEventListener(window, 'resize', this.handleResize);
    this.socket = this.$createWebSocket('ws://xxx');
  }
};
```

**优化效果：**

// utils/chunkUploader.js

export class ChunkUploader {
  constructor(file, options = {}) {
    this.file = file;
    this.chunkSize = options.chunkSize || 2 * 1024 * 1024; // 2MB
    this.concurrency = options.concurrency || 3;
    this.chunks = [];
    this.uploadedChunks = [];
    this.fileId = this.generateFileId();
  }
  
  /**
   * 生成文件 ID
   */
  generateFileId() {
    return `${this.file.name}_${this.file.size}_${this.file.lastModified}`;
  }
  
  /**
   * 创建分片
   */
  createChunks() {
    let start = 0;
    let index = 0;
    
    while (start < this.file.size) {
      const end = Math.min(start + this.chunkSize, this.file.size);
      this.chunks.push({
        index,
        start,
        end,
        blob: this.file.slice(start, end)
      });
      start = end;
      index++;
    }
  }
  
  /**
   * 上传分片
   */
  async uploadChunk(chunk, retryCount = 0) {
    const formData = new FormData();
    formData.append('fileId', this.fileId);
    formData.append('chunk', chunk.blob);
    formData.append('index', chunk.index);
    formData.append('total', this.chunks.length);
    
    try {
      await http.upload({
        url: '/upload/chunk',
        data: formData
      });
      this.uploadedChunks.push(chunk.index);
    } catch (error) {
      if (retryCount < 3) {
        return this.uploadChunk(chunk, retryCount + 1);
      }
      throw error;
    }
  }
  
  /**
   * 上传所有分片
   */
  async upload(onProgress) {
    this.createChunks();
    
    // 检查已上传的分片
    await this.checkUploadedChunks();
    
    // 并发上传
    for (let i = 0; i < this.chunks.length; i += this.concurrency) {
      const pendingChunks = this.chunks
        .slice(i, i + this.concurrency)
        .filter(chunk => !this.uploadedChunks.includes(chunk.index));
      
      await Promise.all(
        pendingChunks.map(chunk => this.uploadChunk(chunk))
      );
      
      // 进度更新
      if (onProgress) {
        const progress = (this.uploadedChunks.length / this.chunks.length) * 100;
        onProgress(progress);
      }
    }
    
    // 合并分片
    return this.mergeChunks();
  }
  
  /**
   * 合并分片
   */
  async mergeChunks() {
    return http.post({
      url: '/upload/merge',
      data: {
        fileId: this.fileId,
        filename: this.file.name,
        total: this.chunks.length
      }
    });
  }
}

// 使用
const uploader = new ChunkUploader(file, {
  chunkSize: 5 * 1024 * 1024,
  concurrency: 5
});

await uploader.upload((progress) => {
  console.log(`上传进度：${progress.toFixed(2)}%`);
});
```

**优化效果：**
- 支持大文件上传

- 失败重试
