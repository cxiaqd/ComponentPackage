2. **分离响应式和非响应式数据**

```javascript
// 优化建议：在组件中
export default {
  data() {
    return {
      // 响应式数据（需要触发视图更新）
      searchResults: [],
      loading: false,
      
      // 非响应式数据（不需要触发视图更新）
      _cachedData: null,
      _requestCache: new Map()
    }
  },
  methods: {
    async fetchResults() {
      // 缓存请求结果（非响应式）
      const cacheKey = JSON.stringify(this.searchParams)
      if (this._requestCache.has(cacheKey)) {
        this._cachedData = this._requestCache.get(cacheKey)
        return
      }
      
      // 响应式数据更新
      const res = await api.search(this.searchParams)
      this.searchResults = res.data
      this._requestCache.set(cacheKey, res.data)
    }
  }
}
```

```javascript
// src/store/common/mutations.js
export default {
  // 添加图片上传历史
  addImage(state, img) {
    if (localStorage.getItem('IMAGE_HISTORY')) {
      state.imageHistory = JSON.parse(localStorage.getItem('IMAGE_HISTORY'))
    } else {
      state.imageHistory = []
    }
    
    if (img) {
      // 问题：直接遍历修改数组
      state.imageHistory.forEach((Img, i) => {
        if (Img.url === img.url) {
          state.imageHistory.splice(i, 1)  // ✅ 使用 splice 可以触发响应式
        }
      })
      state.imageHistory.push(img)  // ✅ 使用 push 可以触发响应式
    }
    
    if (state.imageHistory.length > 5) {
      state.imageHistory.splice(0, 1)  // ✅ 使用 splice 可以触发响应式
    }
    localStorage.setItem('IMAGE_HISTORY', JSON.stringify(state.imageHistory))
  }
}
```

```javascript
      containerHeight: 600

        ...item,
        top: (this.startIndex + index) * this.itemSize

    }

      this.scrollTop = e.target.scrollTop
```

```javascript
// 2. 图片预加载工具
// src/utils/imagePreload.js
export class ImagePreloader {
  constructor() {
    this.cache = new Map()
    this.queue = []
    this.loading = false
  }
  
  async preload(urls) {
    const promises = urls.map(url => this.loadImage(url))
    return Promise.all(promises)
  }
  
  loadImage(url) {
    if (this.cache.has(url)) {
      return Promise.resolve(this.cache.get(url))
    }
    
    return new Promise((resolve, reject) => {
      const img = new Image()
      img.onload = () => {
        this.cache.set(url, img)
        resolve(img)
      }
      img.onerror = reject
      img.src = url
    })
  }
  
  // 预加载搜索结果的前几张图片
  async preloadSearchResults(results, count = 5) {
    const urls = results.slice(0, count).map(item => {
      return this.getOptimizedUrl(item.imageUrl)
    })
    return this.preload(urls)
  }
}
```

```javascript
  constructor() {
```

```javascript
    this.loadingModules = new Map()

    // 检查是否已加载

      return this.loadedModules.get(moduleName)
    }
    
    // 检查是否正在加载
    if (this.loadingModules.has(moduleName)) {
      return this.loadingModules.get(moduleName)
    }
    
    const loadingPromise = this.doLoadModule(moduleName, config)
    this.loadingModules.set(moduleName, loadingPromise)
    
    try {
      const module = await loadingPromise
      this.loadedModules.set(moduleName, module)
      this.loadingModules.delete(moduleName)
      return module
    } catch (error) {
      this.loadingModules.delete(moduleName)
      throw error
    }
  }
  
  async doLoadModule(moduleName, config) {

      // 使用降级方案

        return await config.fallback()
      }
```
