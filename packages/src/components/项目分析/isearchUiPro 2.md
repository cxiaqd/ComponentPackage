// packages/utils/src/business/rightDetailPro/handle-right-detail.js
const handleRightDetail = (param) => {
  let { cardInfo = {}, type, moduleName = "text", rightDetailConfig } = param;

  const rightDetail = {
    rightDetailData: [],
    rightDetailTitle: {},
  };

  // 1. 数据预处理
  if (cardInfo.hitInfo && moduleName === "image") {
    param.cardInfo = cardInfo.hitInfo[0];
    cardInfo = cardInfo.hitInfo[0];
  }

  // 2. 获取配置
  let useRightConfig = RMainConfig;
  if (rightDetailConfig && rightDetailConfig.RMainConfig)
    useRightConfig = rightDetailConfig.RMainConfig;

  const mainConfig = useRightConfig.mainConfig[moduleName];
  const subConfig = useRightConfig.subConfig[type];

  // 3. 处理标题信息
  let titleInfo = {};
  if (mainConfig.hasTitle || (subConfig && subConfig.hasTitle))
    titleInfo = handleRightTitle(param);

  // 4. 模块处理函数映射
  const funList = {
    handleHitField,        // 命中字段
    handleOriginPic,       // 抓拍原图
    handleComparePro,      // 抓拍比对
    handleFeatureCompare,  // 特征比对
    handleVideoPlay,       // 视频回放
    handleSyncCollect,     // 同步采集
    handleMultiCollect,    // 多维关系
    handleMapLocation,     // 地图位置
    handleHumanIdentity,   // 身份推荐
    handleFeatureInfo,     // 特征信息
    handleSimSnap,         // 前后脚通过
    handleBuiltInfo,       // 体态信息
    handleBuiltVideo,      // 体态视频
    handleModuleSlot,      // 模块插槽
  };

  // 5. 组装模块数据
  let modules = mainConfig.modules;
  if (subConfig && subConfig.modules) modules = subConfig.modules;

  const rightDetailData = [];
  modules.map((item) => {
    const funName = `handle${item}`;
    const thisFun = funList[funName];
    const returnData = thisFun(param);
    if (returnData) rightDetailData.push(returnData);
  });

  // 6. 模块分组处理
  const moduleGroup = useRightConfig.moduleGroup;
  const newDetailArray = [];
  Object.values(moduleGroup).map((config) => {
    const newList = [];
    config.map((thisConfig) => {
      const thisItem = rightDetailData.filter((item) =>
        item.trueKey ? item.trueKey === thisConfig : item.key === thisConfig
      )[0];
      if (thisItem) newList.push(thisItem);
    });
    if (newList.length) newDetailArray.push(newList);
  });

  rightDetail.rightDetailData = newDetailArray;
  rightDetail.rightDetailTitle = titleInfo;

  return rightDetail;
};

```javascript
// packages/utils/src/base/throttle.js
export default function throttle(fn, delay = 300) {
  let timer = null;
  let lastTime = 0;

  return function(...args) {
    const now = Date.now();
    const remaining = delay - (now - lastTime);

    if (remaining <= 0 || remaining > delay) {
      if (timer) {
        clearTimeout(timer);
        timer = null;
      }
      lastTime = now;
      fn.apply(this, args);
    } else if (!timer) {
      timer = setTimeout(() => {
        lastTime = Date.now();
        timer = null;
        fn.apply(this, args);
      }, remaining);
    }
  };
}
```

#### 4.2.3 深拷贝优化
```javascript
// packages/utils/src/base/deep-clone.js
export default function deepClone(obj, hash = new WeakMap()) {
  // 处理基本类型
  if (obj === null || typeof obj !== 'object') {
    return obj;
  }

  // 处理日期对象
  if (obj instanceof Date) {
    return new Date(obj);
  }

  // 处理正则表达式
  if (obj instanceof RegExp) {
    return new RegExp(obj);
  }

  // 处理循环引用
  if (hash.has(obj)) {
    return hash.get(obj);
  }

  // 创建新对象
  const cloneObj = Array.isArray(obj) ? [] : {};
  hash.set(obj, cloneObj);

  // 递归拷贝
  for (const key in obj) {
    if (obj.hasOwnProperty(key)) {
      cloneObj[key] = deepClone(obj[key], hash);
    }
  }

  return cloneObj;
}
```

**优化点**：
- 使用 `WeakMap` 处理循环引用
- 避免重复拷贝
- 支持日期、正则等特殊对象
